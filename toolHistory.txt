    if (n != 4) return -1;
    *out = ((uint32_t)buf[0]) |
           ((uint32_t)buf[1] << 8) |
           ((uint32_t)buf[2] << 16) |
           ((uint32_t)buf[3] << 24);
    return 0;
}

// Read a single byte
int read_u8(int fd, uint8_t *out) {
    ssize_t n = read(fd, out, 1);
    return (n == 1) ? 0 : -1;
}
